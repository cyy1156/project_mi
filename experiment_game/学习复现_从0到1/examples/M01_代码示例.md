### 2A. 继承版在解决什么问题？

```text
BaseSampleSource（父类）
    ├── 公共：start / stop / _running
    ├── 抽象：iter_n（子类必须实现）
    └── 可选默认：samples（父类写好，子类一般不用改）

SyntheticSource（子类）     → M01 合成假数据
CytonSource（以后）         → 真机 COM/LSL（M01 先不做）
ReplaySource（以后）       → 从 CSV 回放（调试用）
```

好处：**换数据源只换子类**，后面 M02 的 Buffer 只认「父类接口」，不用改。

### 2B. 父类骨架（示例 · 请自己敲进 `source_base.py`）

下面**故意留空**几处，让你对照注释补全，不要整段粘贴交差。

```python
"""样本源基类 —— 继承版（可选）。"""

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Iterator, Tuple

import numpy as np

from self_learing.channels import FS_HZ, N_CH


class BaseSampleSource(ABC):
    """所有样本源的父类：规定「必须有什么方法」，并复用 start/stop。"""

    def __init__(self, fs: float = FS_HZ, n_ch: int = N_CH) -> None:
        if int(n_ch) != N_CH:
            raise ValueError(f"n_ch 必须是 {N_CH}，收到 {n_ch}")
        self.fs = float(fs)
        self.n_ch = int(n_ch)
        self._running = False

    # ---------- 公共：子类一般不用重写 ----------
    def start(self) -> None:
        """对应原版 AcquisitionFacade.start。"""
        self._running = True
        self._on_start()  # 钩子：子类有需要再 override

    def stop(self) -> None:
        """对应原版 AcquisitionFacade.stop。"""
        self._running = False
        self._on_stop()

    def _on_start(self) -> None:
        """子类可选：打开串口、连 LSL 等。"""
        pass

    def _on_stop(self) -> None:
        """子类可选：关串口、断开 LSL 等。"""
        pass

    # ---------- 抽象：子类必须实现 ----------
    @abstractmethod
    def iter_n(
        self, n: int, start_t: float = 0.0
    ) -> Iterator[Tuple[float, np.ndarray]]:
        """一次产出 n 个 (t, x)，x.shape == (n_ch,)。"""
        raise NotImplementedError  # 抽象方法；子类必须写具体逻辑

    # ---------- 默认实现：多数子类可直接用 ----------
    def samples(self) -> Iterator[Tuple[float, np.ndarray]]:
        """实时风格无限流；测试优先用 iter_n。"""
        if not self._running:
            self.start()
        i = 0
        while self._running:
            # 提示：也可以写成 for t, x in self.iter_n(1, start_t=i/self.fs): yield ...
            t = i / self.fs
            x = self._pull_one_sample()  # 见下面：子类实现「取一个点」
            yield t, x
            i += 1

    @abstractmethod
    def _pull_one_sample(self) -> np.ndarray:
        """子类实现：返回 shape (n_ch,) 的一个样本。"""
        raise NotImplementedError
```

**你要自己补的点（练习）：**

1. `iter_n` 里用 `for i in range(n):` + `yield t, x`（见 2C 子类示例）。  
2. 若觉得 `_pull_one_sample` 太细，可以只在子类写 `iter_n`，父类 `samples` 里用 `while` + 调 `iter_n(1, ...)`（两种风格任选一种）。

### 2C. 子类示例：`SyntheticSource` 怎么继承

```python
"""合成样本泵 —— 继承 BaseSampleSource。"""

from __future__ import annotations

from typing import Iterator, Optional, Tuple

import numpy as np

from self_learing.channels import FS_HZ, N_CH
from self_learing.source_base import BaseSampleSource


class SyntheticSource(BaseSampleSource):
    def __init__(
        self,
        fs: float = FS_HZ,
        n_ch: int = N_CH,
        seed: Optional[int] = 0,
    ) -> None:
        super().__init__(fs=fs, n_ch=n_ch)  # 先让父类检查 n_ch、设 fs
        self._rng = np.random.default_rng(seed)

    def _pull_one_sample(self) -> np.ndarray:
        """父类 samples() 会调这里。"""
        return self._rng.standard_normal(self.n_ch).astype(np.float64)

    def iter_n(
        self, n: int, start_t: float = 0.0
    ) -> Iterator[Tuple[float, np.ndarray]]:
        n = int(n)
        for i in range(n):
            t = float(start_t) + i / self.fs
            x = self._pull_one_sample()
            yield t, x
```

**对照记忆：**

| 关键字 | 作用 |
|--------|------|
| `class SyntheticSource(BaseSampleSource)` | 声明「我是父类的子类」 |
| `super().__init__(...)` | 先跑父类 `__init__`，再写自己的 `_rng` |
| `@abstractmethod` 的方法 | 父类说「你必须实现」；不实现就不能 `SyntheticSource()` |
| `_pull_one_sample` | 只负责「一个点」；`iter_n` 负责「n 个点 + 时间戳」 |

### 2E. 和 Protocol 版对比（懂一行就够）

| | Protocol（原示例 2） | 继承 BaseSampleSource |
|--|---------------------|------------------------|
| 关系 | 「像就行」，不必显式继承 | 必须 `class X(BaseSampleSource)` |
| 公共代码 | 每个类自己写 start/stop | 写在父类，子类 `super()` |
| 漏写方法 | 运行时才可能发现 | 抽象方法漏写，创建类/实例时更容易发现 |
| M01 过关 | 两种都可以 | 两种都可以 |

## 文件 3：`source_synthetic.py`（完整 · 核心）

逐行说明写在注释里，请慢慢读。

```python
"""合成样本泵。

对照原版 AcquisitionFacade：
  - start/stop：开关采集
  - 这里不碰 COM/LSL，只制造假数据给后面 Buffer 用
"""

from __future__ import annotations

from typing import Iterator, Optional, Tuple

import numpy as np

from teach_demo.channels import FS_HZ, N_CH  # 作业里改成 self_learing.channels


class SyntheticSource:
    def __init__(
        self,
        fs: float = FS_HZ,
        n_ch: int = N_CH,
        seed: Optional[int] = 0,
    ) -> None:
        # 强制 8 导，避免后面模型 shape 错乱
        if int(n_ch) != N_CH:
            raise ValueError(f"n_ch 必须是 {N_CH}，收到 {n_ch}")
        self.fs = float(fs)
        self.n_ch = int(n_ch)
        # 固定 seed → 测试可重复
        self._rng = np.random.default_rng(seed)
        self._running = False

    def start(self) -> None:
        """对应原版 Facade.start：开始出数。"""
        self._running = True

    def stop(self) -> None:
        """对应原版 Facade.stop：停止出数。"""
        self._running = False

    def iter_n(
        self, n: int, start_t: float = 0.0
    ) -> Iterator[Tuple[float, np.ndarray]]:
        """一次产出 n 个样本，不 sleep（单测友好）。

        每个元素：
          t: float  秒
          x: np.ndarray shape (8,)
        """
        n = int(n)
        for i in range(n):
            # 第 i 个点的时间：起点 + i/采样率
            t = float(start_t) + i / self.fs
            # 假脑电：标准正态；真机这里会是真实电压
            x = self._rng.standard_normal(self.n_ch).astype(np.float64)
            yield t, x

    def samples(self) -> Iterator[Tuple[float, np.ndarray]]:
        """实时风格无限流。测试请优先用 iter_n。"""
        if not self._running:
            self.start()
        i = 0
        while self._running:
            t = i / self.fs
            x = self._rng.standard_normal(self.n_ch).astype(np.float64)
            yield t, x
            i += 1
            # 真·实时可在这里 time.sleep(1/self.fs)；教学示例不写 sleep
```

## 文件 4：`tests/test_synthetic_source.py`（完整）

```python
import sys
from pathlib import Path

# 让 Python 找到 src/ 下的包
sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))

import numpy as np

from teach_demo.channels import CHANNEL_NAMES, N_CH  # 作业：self_learing.channels
from teach_demo.source_synthetic import SyntheticSource


def test_iter_n_count_and_shape():
    src = SyntheticSource(fs=250.0, n_ch=8, seed=0)
    xs = list(src.iter_n(250))
    assert len(xs) == 250
    t0, x0 = xs[0]
    assert isinstance(t0, float)
    assert isinstance(x0, np.ndarray)
    assert x0.shape == (8,)
    assert x0.dtype == np.float64


def test_time_spacing():
    src = SyntheticSource(fs=250.0, seed=1)
    xs = list(src.iter_n(3, start_t=1.0))
    assert abs(xs[0][0] - 1.0) < 1e-9
    assert abs(xs[1][0] - (1.0 + 1 / 250.0)) < 1e-9


def test_channel_names_match_openbmi_order():
    # 与原版 DEVICE_CHANNEL_LABELS 一致
    assert CHANNEL_NAMES == [
        "FC3", "C3", "CP3", "CZ", "CPZ", "FC4", "C4", "CP4"
    ]
    assert N_CH == 8


def test_start_stop_flags():
    src = SyntheticSource()
    src.start()
    assert src._running is True
    src.stop()
    assert src._running is False
```
