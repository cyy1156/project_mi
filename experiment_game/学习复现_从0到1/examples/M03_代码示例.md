# M03 教学代码示例 · events + alignment

> **对照**：对照场 `events.jsonl` 字段名；`markers.py`；`alignment.py`。

## 三句话

1. 每个实验事件写成 jsonl 一行。  
2. Marker 先只写日志（不必真 LSL）。  
3. alignment：事件时间必须落在 EEG 时间范围内。

## `events.py`

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, List


class EventLogger:
    def __init__(self, path: Path) -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._f = self.path.open("a", encoding="utf-8")

    def append(self, name: str, t: float, **fields: Any) -> None:
        row: Dict[str, Any] = {"name": str(name), "t": float(t)}
        row.update(fields)
        self._f.write(json.dumps(row, ensure_ascii=False) + "\n")
        self._f.flush()

    def close(self) -> None:
        self._f.close()


def read_events(path: Path) -> List[Dict[str, Any]]:
    rows = []
    for line in Path(path).read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if line:
            rows.append(json.loads(line))
    return rows
```

> 若你对照场里字段叫 `event` 而不是 `name`，把键改成与对照场一致。

## `alignment.py`

```python
from __future__ import annotations

from typing import Any, Dict, List


def verify_events(
    events: List[Dict[str, Any]], t0: float, t1: float
) -> Dict[str, Any]:
    checks = []
    ok = True
    for e in events:
        t = float(e["t"])
        in_range = (t0 <= t <= t1)
        checks.append({"name": e.get("name"), "t": t, "in_range": in_range})
        ok = ok and in_range
    return {"passed": ok, "checks": checks}
```

## 测试摘要

```python
def test_roundtrip(tmp_path):
    from teach_demo.events import EventLogger, read_events
    p = tmp_path / "events.jsonl"
    lg = EventLogger(p)
    lg.append("rest_start", 0.0, label="Left")
    lg.append("mi_start", 6.0, label="Left")
    lg.close()
    rows = read_events(p)
    assert [r["name"] for r in rows] == ["rest_start", "mi_start"]

def test_alignment_fail():
    from teach_demo.alignment import verify_events
    ev = [{"name": "x", "t": 999.0}]
    rep = verify_events(ev, 0.0, 10.0)
    assert rep["passed"] is False
```
