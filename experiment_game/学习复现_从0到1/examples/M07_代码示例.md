## 规则

1. 每窗有 `p_three`（长度 3）。  
2. 窗 i：对 `p[max(0,i-2):i+1]` 做平均 → argmax → `pred[i]`（因果，lookback=2）。  
3. 全部 `pred` 多数票 = 试次预测。  
4. 与真值比：Left/Right 对 +1；Rest 对 +0.5。

## `readout.py`

```python
from __future__ import annotations

from typing import List, Sequence

import numpy as np


def causal_smooth_preds(
    probs: Sequence[np.ndarray], lookback: int = 2
) -> List[int]:
    """返回每窗平滑后的 pred（int）。

    更完整的原版返回 dict 列表；教学版先返回 pred 序列。
    """
    probs = [np.asarray(p, dtype=np.float64).reshape(3) for p in probs]
    preds: List[int] = []
    for i in range(len(probs)):
        lo = max(0, i - lookback)
        mean_p = np.mean(probs[lo : i + 1], axis=0)
        preds.append(int(mean_p.argmax()))
    return preds


def majority_vote(preds: Sequence[int]) -> int:
    vals, counts = np.unique(np.asarray(preds, dtype=int), return_counts=True)
    # 票数最多；平票时取 vals 中第一个最大票（原版可能有更细规则，请对照）
    return int(vals[np.argmax(counts)])
```

## `scoring.py`

```python
from __future__ import annotations

from typing import Any, Dict, List

import numpy as np

from teach_demo.readout import causal_smooth_preds, majority_vote

# 示例映射：请与你的模型输出类索引统一，并写进 README
IDX_TO_NAME = {0: "Rest", 1: "Left", 2: "Right"}
NAME_TO_IDX = {v: k for k, v in IDX_TO_NAME.items()}


class MiTrialTracker:
    def __init__(self, true_label: str) -> None:
        self.true_label = true_label
        self.probs: List[np.ndarray] = []

    def add_window(self, p_three) -> None:
        self.probs.append(np.asarray(p_three, dtype=np.float64).reshape(3))

    def finalize(self) -> Dict[str, Any]:
        if not self.probs:
            return {"pred": None, "score": 0.0, "ok": False}
        preds = causal_smooth_preds(self.probs, lookback=2)
        pred_idx = majority_vote(preds)
        pred_name = IDX_TO_NAME[pred_idx]
        ok = pred_name == self.true_label
        if not ok:
            score = 0.0
        elif self.true_label == "Rest":
            score = 0.5
        else:
            score = 1.0
        return {
            "pred": pred_name,
            "pred_idx": pred_idx,
            "window_preds": preds,
            "ok": ok,
            "score": score,
        }
```

def test_causal_lookback2_hand():
    import numpy as np
    from teach_demo.readout import causal_smooth_preds
    # 三窗：分别明显偏向 0, 1, 0
    p0 = np.array([0.8, 0.1, 0.1])
    p1 = np.array([0.1, 0.8, 0.1])
    p2 = np.array([0.8, 0.1, 0.1])
    preds = causal_smooth_preds([p0, p1, p2], lookback=2)
    # i=0: 只有 p0 → 0
    # i=1: mean(p0,p1) → 看谁大
    # i=2: mean(p0,p1,p2)
    assert preds[0] == 0
    assert len(preds) == 3


def test_rest_half_point():
    from teach_demo.scoring import MiTrialTracker
    import numpy as np
    tr = MiTrialTracker("Rest")
    # 三窗都偏向 Rest=0
    for _ in range(3):
        tr.add_window([0.9, 0.05, 0.05])
    out = tr.finalize()
    assert out["ok"] is True
    assert out["score"] == 0.5
