# M09 教学代码示例 · 落盘产品化

> **对照**：对照场 `run_config.json` / `session.meta.json` / `alignment/verify_report.json`；原版 `session_layout.py`、`subject_registry.py`、`alignment.py`。  
> **作业路径**（自己敲）：
>
> ```text
> self_learing/src/self_learing/
>   session_layout.py     ← write_run_config, write_session_meta, finalize_session_dir
>   subject_registry.py   ← ensure_subject
>   alignment.py          ← 在 M03 基础上增强（本文件给出增强版）
> self_learing/tests/
>   test_session_layout.py
> ```

## 三句话

1. 参数必须落盘成 `run_config.json`，不能只活在内存。  
2. `session.meta.json` 要有 `phase_mode` 与 `complete`（给 Leave-Next 过滤用）。  
3. alignment 报告写到 `alignment/verify_report.json`。

## `session_layout.py` → `self_learing/src/self_learing/session_layout.py`

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, List, Optional


def write_run_config(out_dir: Path, cfg: Dict[str, Any]) -> Path:
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    path = out_dir / "run_config.json"
    path.write_text(json.dumps(cfg, ensure_ascii=False, indent=2), encoding="utf-8")
    return path


def write_session_meta(
    out_dir: Path,
    *,
    subject: str,
    phase_mode: str = "v3_session",
    complete: bool = True,
    extra: Optional[Dict[str, Any]] = None,
) -> Path:
    out_dir = Path(out_dir)
    meta: Dict[str, Any] = {
        "subject": subject,
        "phase_mode": phase_mode,
        "complete": bool(complete),
    }
    if extra:
        meta.update(extra)
    path = out_dir / "session.meta.json"
    path.write_text(json.dumps(meta, ensure_ascii=False, indent=2), encoding="utf-8")
    return path


def write_alignment_report(out_dir: Path, report: Dict[str, Any]) -> Path:
    d = Path(out_dir) / "alignment"
    d.mkdir(parents=True, exist_ok=True)
    path = d / "verify_report.json"
    path.write_text(json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8")
    return path


def finalize_session_dir(
    out_dir: Path,
    *,
    subject: str,
    cfg: Dict[str, Any],
    events: List[Dict[str, Any]],
    complete: bool = True,
    eeg_t0: Optional[float] = None,
    eeg_t1: Optional[float] = None,
) -> Dict[str, Any]:
    """一次收尾：run_config + meta + alignment。"""
    from self_learing.alignment import verify_events

    write_run_config(out_dir, cfg)
    write_session_meta(out_dir, subject=subject, complete=complete)

    if eeg_t0 is None or eeg_t1 is None:
        ts = [float(e["t"]) for e in events] or [0.0]
        eeg_t0, eeg_t1 = min(ts), max(ts)
    report = verify_events(events, float(eeg_t0), float(eeg_t1))
    write_alignment_report(out_dir, report)
    return report
```

`cfg` 至少含：`rest_s/prep_s/cue_s/imagine_s/iti_s`、`fs`、`channels`、`readout_mode`（对照你对照场键名补齐）。

## `subject_registry.py` → `self_learing/src/self_learing/subject_registry.py`

```python
from __future__ import annotations

from pathlib import Path


def ensure_subject(root: Path, subject_id: str) -> Path:
    """在 data 根下准备 subjects/<id>/… 骨架。"""
    root = Path(root)
    p = root / "subjects" / str(subject_id)
    (p / "models" / "current").mkdir(parents=True, exist_ok=True)
    (p / "models" / "ft_runs").mkdir(parents=True, exist_ok=True)
    (p / "sessions").mkdir(parents=True, exist_ok=True)
    return p
```

建议 `root = self_learing/data`（或你 README 声明的路径）。

## 增强 `alignment.py`（覆盖/合并进 M03 文件）

在 M03 `verify_events` 旁增加：

```python
def verify_events_rich(
    events: List[Dict[str, Any]], t0: float, t1: float
) -> Dict[str, Any]:
    base = verify_events(events, t0, t1)
    names = [str(e.get("name")) for e in events]
    has_mi = any(n == "mi_start" for n in names)
    base["checks"].append({"name": "has_mi_start", "ok": has_mi})
    base["passed"] = bool(base["passed"] and has_mi)
    return base
```

## `tests/test_session_layout.py`

```python
from __future__ import annotations

import json
from pathlib import Path

from self_learing.session_layout import (
    finalize_session_dir,
    write_run_config,
    write_session_meta,
)
from self_learing.subject_registry import ensure_subject


def test_run_config_has_imagine(tmp_path: Path) -> None:
    cfg = {"imagine_s": 4.0, "fs": 250}
    write_run_config(tmp_path, cfg)
    data = json.loads((tmp_path / "run_config.json").read_text(encoding="utf-8"))
    assert data["imagine_s"] == 4.0


def test_abort_complete_false(tmp_path: Path) -> None:
    write_session_meta(tmp_path, subject="demo", complete=False)
    meta = json.loads((tmp_path / "session.meta.json").read_text(encoding="utf-8"))
    assert meta["complete"] is False
    assert meta["phase_mode"] == "v3_session"


def test_finalize_and_subject(tmp_path: Path) -> None:
    events = [{"name": "mi_start", "t": 1.0}, {"name": "mi_end", "t": 5.0}]
    rep = finalize_session_dir(
        tmp_path / "sess",
        subject="syj",
        cfg={"imagine_s": 4.0},
        events=events,
        complete=True,
    )
    assert (tmp_path / "sess" / "alignment" / "verify_report.json").is_file()
    assert "passed" in rep
    sub = ensure_subject(tmp_path / "data", "syj")
    assert (sub / "models" / "current").is_dir()
```
