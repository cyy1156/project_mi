## 主循环长什么样

把下面保存为 `self_learing/src/self_learing/session_v3.py`（import 用 `self_learing`）：

```python
from __future__ import annotations

import json
from pathlib import Path

from self_learing.alignment import verify_events
from self_learing.csv_recorder import CsvRecorder  # 若文件名是 csv_recoder，改 import
from self_learing.channels import CHANNEL_NAMES
from self_learing.eeg_bus import EegBus
from self_learing.events import EventLogger, read_events
from self_learing.infer_fake import FakeModel
from self_learing.inference_service import InferenceService
from self_learing.live_capture import LiveCapture
from self_learing.ring_buffer import RingBuffer
from self_learing.scoring import MiTrialTracker
from self_learing.source_synthetic import SyntheticSource
from self_learing.timing import TrialTiming, load_timing
from self_learing.trial_fsm import run_trial
from self_learing.windowing import iter_window_specs
# 真实系统里：窗尾时刻到了再从 Buffer 切片 → preprocess → judge
# 教学简化：每个 trial 用 Fake 直接塞若干窗概率


def run_v3_session_demo(out_dir: Path, timing: TrialTiming, labels: list[str]) -> None:
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)

    # 1) 落盘配置快照
    (out_dir / "run_config.json").write_text(
        json.dumps(
            {
                "rest_s": timing.rest_s,
                "prep_s": timing.prep_s,
                "cue_s": timing.cue_s,
                "imagine_s": timing.imagine_s,
                "iti_s": timing.iti_s,
                "fs": 250,
                "channels": CHANNEL_NAMES,
                "readout_mode": "fake_f5",
            },
            ensure_ascii=False,
            indent=2,
        ),
        encoding="utf-8",
    )

    logger = EventLogger(out_dir / "events.jsonl")
    src = SyntheticSource(seed=0)
    buf = RingBuffer()
    bus = EegBus()
    rec = CsvRecorder(out_dir / "eeg.csv", CHANNEL_NAMES)
    bus.subscribe(rec.on_sample)
    capture = LiveCapture(src, buf, bus)

    # 教学：整场先灌一段假 EEG（真系统是边跑 trial 边采）
    capture.run_n_samples(250 * 60)  # 约 60s，按你试次数调整

    svc = InferenceService(FakeModel((0.2, 0.6, 0.2)))
    total_score = 0.0
    t = 0.0

    for lab in labels:
        t = run_trial(lab, timing, logger.append, t0=t)
        # 简化：不按真实窗尾取缓冲，直接加 11 个假窗
        tracker = MiTrialTracker(lab)
        for _ in iter_window_specs(4.0, 3.0, 0.1):
            import numpy as np
            out = svc.judge_window(np.zeros((8, 750)))
            tracker.add_window(out["p_three"])
        result = tracker.finalize()
        total_score += float(result["score"])
        logger.append("trial_score", t, label=lab, **{k: result[k] for k in ("pred", "score", "ok")})

    rec.close()
    logger.close()

    events = read_events(out_dir / "events.jsonl")
    # 简化：用事件时间范围做 verify
    ts = [float(e["t"]) for e in events]
    report = verify_events(events, min(ts), max(ts))
    (out_dir / "alignment").mkdir(exist_ok=True)
    (out_dir / "alignment" / "verify_report.json").write_text(
        json.dumps(report, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    (out_dir / "session.meta.json").write_text(
        json.dumps(
            {"phase_mode": "v3_session", "complete": True, "total_score": total_score},
            ensure_ascii=False,
            indent=2,
        ),
        encoding="utf-8",
    )
```

## 你真正实现时要升级的点

1. **边 trial 边采**：`run_trial` 的每个阶段与实时 `iter` 对齐（或线程泵）。  
2. **按 `mi_start` + 窗尾**从 `RingBuffer` 切片，再 preprocess，再 judge。  
3. 判定时刻列表对照 `openbmi_align_config`。  

先把「目录里能出现 events/run_config/eeg/verify」跑通，再升级真实取窗。

## CLI 入口 → `self_learing/tools/run_synthetic_session.py`

```python
from __future__ import annotations

import sys
from pathlib import Path

# 允许直接 python tools/run_synthetic_session.py
ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))

from self_learing.session_v3 import run_v3_session_demo
from self_learing.timing import load_timing

if __name__ == "__main__":
    timing = load_timing(ROOT / "config" / "protocol.yaml")
    out = ROOT / "data" / "sessions" / "demo_run"
    run_v3_session_demo(out, timing, labels=["Left", "Right", "Rest"])
    print("wrote", out)
```

## 冒烟测试 → `self_learing/tests/test_session_smoke.py`

```python
from __future__ import annotations

from pathlib import Path

from self_learing.session_v3 import run_v3_session_demo
from self_learing.timing import TrialTiming


def test_session_writes_files(tmp_path: Path) -> None:
    timing = TrialTiming(0.1, 0.1, 0.1, 0.2, 0.1)  # 缩短，加快测试
    out = tmp_path / "sess"
    run_v3_session_demo(out, timing, labels=["Left"])
    assert (out / "events.jsonl").is_file()
    assert (out / "run_config.json").is_file()
    assert (out / "session.meta.json").is_file()
```
