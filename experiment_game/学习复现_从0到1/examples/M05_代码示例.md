2. 每窗采样 250×3=**750** 点，形状 `(8, 750)`。

## `windowing.py` → `self_learing/src/self_learing/windowing.py`

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import List

import numpy as np


@dataclass
class WindowSpec:
    t_start: float  # 相对锚点（秒）
    t_end: float


def iter_window_specs(
    task_s: float = 4.0, win_s: float = 3.0, hop_s: float = 0.1
) -> List[WindowSpec]:
    specs: List[WindowSpec] = []
    t0 = 0.0
    # 允许微小浮点误差
    while t0 + win_s <= task_s + 1e-9:
        specs.append(WindowSpec(t0, t0 + win_s))
        t0 += hop_s
    return specs


def slice_window_8xT(
    data_tx8: np.ndarray,
    fs: float,
    anchor_t: float,
    spec: WindowSpec,
) -> np.ndarray:
    """data_tx8: shape (T, 8)，按绝对时间从 anchor 起切。

    返回 (8, n_samp)。
    """
    data = np.asarray(data_tx8, dtype=np.float64)
    assert data.ndim == 2 and data.shape[1] == 8
    i0 = int(round((anchor_t + spec.t_start) * fs))
    i1 = int(round((anchor_t + spec.t_end) * fs))
    seg = data[i0:i1]  # (n, 8)
    # 转成 (8, n)
    return seg.T.copy()
```

## `preprocess.py` → `self_learing/src/self_learing/preprocess.py`

```python
from __future__ import annotations

import numpy as np


def zscore_per_channel(seg_8xT: np.ndarray, eps: float = 1e-8) -> np.ndarray:
    """对照 core.windowing.trial_zscore：每行（通道）减均值除标准差。"""
    x = np.asarray(seg_8xT, dtype=np.float64)
    mean = x.mean(axis=1, keepdims=True)
    std = x.std(axis=1, keepdims=True)
    return (x - mean) / (std + eps)


def baseline_correct(
    seg_8xT: np.ndarray, baseline_8xTb: np.ndarray
) -> np.ndarray:
    """段级：每通道减去 baseline 时段均值。"""
    base = baseline_8xTb.mean(axis=1, keepdims=True)
    return np.asarray(seg_8xT, dtype=np.float64) - base
```

## `tests/test_windowing.py` → `self_learing/tests/test_windowing.py`

```python
from __future__ import annotations

import numpy as np

from self_learing.preprocess import zscore_per_channel
from self_learing.windowing import WindowSpec, iter_window_specs, slice_window_8xT


def test_n_windows() -> None:
    specs = iter_window_specs(4.0, 3.0, 0.1)
    # 若与 n_windows_3s_hop100(4.0) 不一致，改成原版返回值
    assert len(specs) == 11
    assert abs(specs[0].t_start - 0.0) < 1e-9
    assert abs(specs[0].t_end - 3.0) < 1e-9


def test_shape_750() -> None:
    fs = 250.0
    T = int(10 * fs)
    data = np.random.randn(T, 8)
    spec = WindowSpec(0.0, 3.0)
    win = slice_window_8xT(data, fs, anchor_t=1.0, spec=spec)
    assert win.shape == (8, 750)
    z = zscore_per_channel(win)
    assert z.shape == (8, 750)
```

> 数据从 `RingBuffer.get_latest` 取出后一般是 `(n, 8)`，再交给 `slice_window_8xT`。
