# M02 教学代码示例 · RingBuffer + Bus + CSV

> **对照原版**：`runtime/eeg_bus.py`（`EEGBus`）、`inference_v2.RingBuffer.push`、`live_capture.py`、`csv_recorder.py`。  
> 作业包名请用 `self_learing`，示例用 `teach_demo`。

## 三句话

1. Bus：一个人 `publish`，所有订阅者都收到。  
2. Buffer：把样本存进环形队列。  
3. CSV：作为订阅者，把每个点写成文件行。

## `eeg_bus.py`

```python
from __future__ import annotations

from typing import Callable, List

import numpy as np

# 订阅者：收到一个样本就调用
Subscriber = Callable[[float, np.ndarray], None]


class EegBus:
    def __init__(self) -> None:
        self._subs: List[Subscriber] = []

    def subscribe(self, fn: Subscriber) -> None:
        self._subs.append(fn)

    def publish(self, t: float, x: np.ndarray) -> None:
        # 拷贝列表，避免订阅中途修改炸掉
        for fn in list(self._subs):
            fn(float(t), np.asarray(x, dtype=np.float64))
```

## `ring_buffer.py`

```python
from __future__ import annotations

from collections import deque
from typing import Deque, List, Tuple

import numpy as np


class RingBuffer:
    def __init__(self, n_ch: int = 8, fs: float = 250.0, capacity_s: float = 30.0) -> None:
        self.n_ch = int(n_ch)
        self.fs = float(fs)
        self.maxlen = max(1, int(capacity_s * fs))
        # 每个元素：(t, x) ；maxlen 满了自动丢最旧
        self._q: Deque[Tuple[float, np.ndarray]] = deque(maxlen=self.maxlen)

    def push(self, t: float, x: np.ndarray) -> None:
        x = np.asarray(x, dtype=np.float64).reshape(-1)
        if x.shape[0] != self.n_ch:
            raise ValueError(f"期望 {self.n_ch} 导，收到 {x.shape}")
        self._q.append((float(t), x.copy()))

    def __len__(self) -> int:
        return len(self._q)

    def get_latest(self, n: int) -> Tuple[np.ndarray, np.ndarray]:
        """返回最近 n 个：times shape (n,)；data shape (n, 8)。"""
        n = int(n)
        if n <= 0 or len(self._q) < n:
            raise ValueError("缓冲不够")
        items = list(self._q)[-n:]
        times = np.array([t for t, _ in items], dtype=np.float64)
        data = np.stack([x for _, x in items], axis=0)  # (n, 8)
        return times, data
```

## `csv_recorder.py`

```python
from __future__ import annotations

from pathlib import Path
from typing import List, TextIO

import numpy as np


class CsvRecorder:
    def __init__(self, path: Path, channel_names: List[str]) -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.channel_names = list(channel_names)
        self._f: TextIO = self.path.open("w", encoding="utf-8", newline="\n")
        header = ["t"] + self.channel_names
        self._f.write(",".join(header) + "\n")

    def on_sample(self, t: float, x: np.ndarray) -> None:
        x = np.asarray(x, dtype=np.float64).reshape(-1)
        cols = [f"{float(t):.6f}"] + [f"{float(v):.8f}" for v in x]
        self._f.write(",".join(cols) + "\n")

    def close(self) -> None:
        self._f.flush()
        self._f.close()
```

## `live_capture.py`（组装）

```python
from __future__ import annotations


class LiveCapture:
    """Source → Buffer.push + Bus.publish（Recorder 已订阅 Bus）。"""

    def __init__(self, source, buffer, bus) -> None:
        self.source = source
        self.buffer = buffer
        self.bus = bus

    def run_n_samples(self, n: int) -> None:
        for t, x in self.source.iter_n(n):
            self.buffer.push(t, x)
            self.bus.publish(t, x)
```

## `tests/test_bus_and_csv.py`

```python
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1] / "src"))

import numpy as np

from teach_demo.channels import CHANNEL_NAMES
from teach_demo.csv_recorder import CsvRecorder
from teach_demo.eeg_bus import EegBus
from teach_demo.live_capture import LiveCapture
from teach_demo.ring_buffer import RingBuffer
from teach_demo.source_synthetic import SyntheticSource


def test_fanout_same_values():
    bus = EegBus()
    a, b = [], []
    bus.subscribe(lambda t, x: a.append((t, x.copy())))
    bus.subscribe(lambda t, x: b.append((t, x.copy())))
    x = np.arange(8.0)
    bus.publish(0.0, x)
    assert len(a) == 1 and len(b) == 1
    assert np.allclose(a[0][1], b[0][1])


def test_csv_500_rows(tmp_path):
    src = SyntheticSource(seed=0)
    buf = RingBuffer()
    bus = EegBus()
    rec = CsvRecorder(tmp_path / "eeg.csv", CHANNEL_NAMES)
    bus.subscribe(rec.on_sample)
    LiveCapture(src, buf, bus).run_n_samples(500)
    rec.close()
    lines = (tmp_path / "eeg.csv").read_text(encoding="utf-8").strip().splitlines()
    assert lines[0].startswith("t,FC3")
    assert len(lines) == 501  # 1 header + 500
    assert len(buf) == 500
```
