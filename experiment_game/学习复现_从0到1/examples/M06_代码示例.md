## 三句话

1. FakeModel：不管输入，返回你注入的 `p_three`。  
2. 温度：`p**(1/T)` 再归一化。  
3. 多成员：温度后加权平均，再归一化。

```python
from __future__ import annotations

import numpy as np


class FakeModel:
    def __init__(self, p_three=(0.5, 0.3, 0.2)) -> None:
        p = np.asarray(p_three, dtype=np.float64).reshape(3)
        p = p / p.sum()
        self.p = p

    def predict_proba(self, win_8x750: np.ndarray) -> np.ndarray:
        _ = win_8x750  # 故意不用，方便测 F5
        return self.p.copy()
```

## `infer_fuse.py`（与原版公式同构的教学版）

```python
from __future__ import annotations

from typing import Sequence

import numpy as np


def apply_temperature_probs(probs: np.ndarray, temperature: float) -> np.ndarray:
    """对照 adapt_engine.e1f.apply_temperature_probs。"""
    p = np.asarray(probs, dtype=np.float64)
    T = float(temperature)
    if T <= 0:
        raise ValueError("temperature 必须 > 0")
    # 常见实现：对概率做 1/T 幂再归一（原版请打开函数核对细节）
    q = np.power(np.clip(p, 1e-12, 1.0), 1.0 / T)
    return q / q.sum(axis=-1, keepdims=True)


def fuse_member_probs(
    probs_list: Sequence[np.ndarray],
    temperatures: Sequence[float],
    weights: Sequence[float],
) -> np.ndarray:
    """对照 fuse_member_probs：先各自温度，再加权平均。"""
    assert len(probs_list) == len(temperatures) == len(weights)
    acc = np.zeros(3, dtype=np.float64)
    wsum = 0.0
    for p, T, w in zip(probs_list, temperatures, weights):
        pt = apply_temperature_probs(p, T)
        acc += float(w) * pt
        wsum += float(w)
    out = acc / max(wsum, 1e-12)
    return out / out.sum()
```

## `inference_service.py`（最小调度）

```python
from __future__ import annotations

from typing import Any, Dict

import numpy as np


class InferenceService:
    def __init__(self, model) -> None:
        self.model = model

    def judge_window(self, win_8x750: np.ndarray) -> Dict[str, Any]:
        p = np.asarray(self.model.predict_proba(win_8x750), dtype=np.float64)
        p = p / p.sum()
        pred = int(p.argmax())
        return {"pred": pred, "p_three": p}
```

标签约定示例：`0=Rest, 1=Left, 2=Right`（以你项目为准写进 README）。
