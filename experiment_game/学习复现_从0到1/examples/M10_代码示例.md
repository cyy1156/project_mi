# M10 教学代码示例 · Leave-Next 调度

> **对照**：`docs/框架冻结确认_*.md`（F7/F8/F9/F11）；`tools/run_leave_next_e1f_task_ramp.py` 的 `RAMP_*`；`pipeline/finetune.py`（只读签名）。  
> **作业路径**：
>
> ```text
> self_learing/src/self_learing/
>   ft_filters.py       ← is_trainable_session
>   leave_next.py       ← build_ramp, fake_finetune
> self_learing/tools/
>   run_leave_next.py
> self_learing/tests/
>   test_leave_next.py
> ```
>
> 真梯度可复用原版 `pipeline/finetune.py`，须在 README 声明；**调度必须自写**。

## 三句话

1. 只训 `phase_mode==v3_session` 且 `complete==True` 的场。  
2. Leave-Next：用前 i 场训，hold 第 i+1 场。  
3. 先 FakeFT 打通 summary，再换真 finetune。

## `ft_filters.py` → `self_learing/src/self_learing/ft_filters.py`

```python
from __future__ import annotations

from typing import Any, Dict


def is_trainable_session(meta: Dict[str, Any]) -> bool:
    if meta.get("phase_mode") != "v3_session":
        return False
    if meta.get("complete") is False:
        return False
    return True
```

## `leave_next.py` → `self_learing/src/self_learing/leave_next.py`

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Any, Dict, List, Tuple


RampStep = Tuple[List[str], str, bool]  # (train_keys, hold_key, use_replay)


def build_ramp(session_keys: List[str]) -> List[RampStep]:
    """对照原版 RAMP_W：keys=[w01..w06] → 5 档。

    use_replay：示例前 2 档 True（i<3）；请打开原版 RAMP_* 把布尔抄准。
    """
    keys = list(session_keys)
    out: List[RampStep] = []
    for i in range(1, len(keys)):
        train = keys[:i]
        hold = keys[i]
        use_replay = i < 3
        out.append((train, hold, use_replay))
    return out


def fake_finetune(
    train_dirs: List[Path], hold_dir: Path, out_dir: Path
) -> Dict[str, Any]:
    """调度打通用：不跑 GPU。"""
    out_dir = Path(out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    summary = {
        "release_pass": False,
        "heldout_acc": 0.0,
        "n_train_sessions": len(train_dirs),
        "hold": str(hold_dir),
        "note": "fake",
    }
    (out_dir / "summary.json").write_text(
        json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    return summary


def load_session_meta(session_dir: Path) -> Dict[str, Any]:
    p = Path(session_dir) / "session.meta.json"
    return json.loads(p.read_text(encoding="utf-8"))
```

## `tools/run_leave_next.py` → `self_learing/tools/run_leave_next.py`

```python
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))

from self_learing.ft_filters import is_trainable_session
from self_learing.leave_next import build_ramp, fake_finetune, load_session_meta


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--subject-root", type=Path, required=True, help="…/subjects/<id>")
    ap.add_argument("--out", type=Path, required=True, help="ft_runs 输出目录")
    args = ap.parse_args()

    sess_root = args.subject_root / "sessions"
    # 约定：每个子目录一场；key=目录名
    candidates = sorted([p for p in sess_root.iterdir() if p.is_dir()])
    keys: list[str] = []
    key_to_dir: dict[str, Path] = {}
    for p in candidates:
        meta = load_session_meta(p)
        if is_trainable_session(meta):
            keys.append(p.name)
            key_to_dir[p.name] = p

    ramp = build_ramp(keys)
    args.out.mkdir(parents=True, exist_ok=True)
    rows = []
    for i, (train_keys, hold_key, use_replay) in enumerate(ramp, start=1):
        train_dirs = [key_to_dir[k] for k in train_keys]
        hold_dir = key_to_dir[hold_key]
        step_out = args.out / f"R{i}_{hold_key}"
        summary = fake_finetune(train_dirs, hold_dir, step_out)
        summary["use_replay"] = use_replay
        summary["train_keys"] = train_keys
        rows.append(summary)
        (step_out / "summary.json").write_text(
            json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8"
        )

    (args.out / "leave_next_overview.json").write_text(
        json.dumps(rows, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    print("steps", len(rows), "->", args.out)


if __name__ == "__main__":
    main()
```

## `tests/test_leave_next.py`

```python
from __future__ import annotations

from self_learing.ft_filters import is_trainable_session
from self_learing.leave_next import build_ramp


def test_filter_rejects_v4_and_incomplete() -> None:
    assert is_trainable_session({"phase_mode": "v3_session", "complete": True})
    assert not is_trainable_session({"phase_mode": "v4_probe", "complete": True})
    assert not is_trainable_session({"phase_mode": "v3_session", "complete": False})


def test_ramp_six_sessions() -> None:
    keys = [f"w0{i}" for i in range(1, 7)]
    ramp = build_ramp(keys)
    assert len(ramp) == 5
    assert ramp[0][0] == ["w01"]
    assert ramp[0][1] == "w02"
    assert ramp[-1][1] == "w06"
```
